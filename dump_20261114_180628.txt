----- FILE: infra/postgres/bootstrap.sql -----
-- Идемпотентно: роли создаются, если их нет; пароли обновляются всегда.
-- Порядок: роли -> пароли -> \connect -> схемы -> владельцы -> гранты -> таймауты.

DO $$
BEGIN
    IF NOT EXISTS (SELECT FROM pg_roles WHERE rolname = 'parser_writer')  THEN CREATE ROLE parser_writer  LOGIN; END IF;
    IF NOT EXISTS (SELECT FROM pg_roles WHERE rolname = 'backend_writer') THEN CREATE ROLE backend_writer LOGIN; END IF;
    IF NOT EXISTS (SELECT FROM pg_roles WHERE rolname = 'matcher_writer') THEN CREATE ROLE matcher_writer LOGIN; END IF;
    IF NOT EXISTS (SELECT FROM pg_roles WHERE rolname = 'catalog_ro')     THEN CREATE ROLE catalog_ro     NOLOGIN; END IF;
END $$;

ALTER ROLE parser_writer  WITH PASSWORD :'parser_pw';
ALTER ROLE backend_writer WITH PASSWORD :'backend_pw';
ALTER ROLE matcher_writer WITH PASSWORD :'matcher_pw';

\connect :"app_db"

CREATE EXTENSION IF NOT EXISTS pg_trgm;
CREATE EXTENSION IF NOT EXISTS vector;

CREATE SCHEMA IF NOT EXISTS parser;
CREATE SCHEMA IF NOT EXISTS catalog;
CREATE SCHEMA IF NOT EXISTS core;
CREATE SCHEMA IF NOT EXISTS matcher;

ALTER SCHEMA parser  OWNER TO parser_writer;
ALTER SCHEMA catalog OWNER TO parser_writer;
ALTER SCHEMA core    OWNER TO backend_writer;
ALTER SCHEMA matcher OWNER TO matcher_writer;


GRANT USAGE ON SCHEMA catalog TO catalog_ro;
GRANT SELECT ON ALL TABLES    IN SCHEMA catalog TO catalog_ro;
GRANT SELECT ON ALL SEQUENCES IN SCHEMA catalog TO catalog_ro;

ALTER DEFAULT PRIVILEGES FOR ROLE parser_writer IN SCHEMA catalog
    GRANT SELECT ON TABLES TO catalog_ro;

GRANT catalog_ro TO backend_writer, matcher_writer;

REVOKE ALL ON SCHEMA parser, core, matcher FROM PUBLIC;
REVOKE CREATE ON SCHEMA public FROM PUBLIC;

ALTER ROLE matcher_writer SET statement_timeout = '5s';
ALTER ROLE backend_writer SET statement_timeout = '5s';
ALTER ROLE parser_writer  SET statement_timeout = '60s';
ALTER ROLE matcher_writer SET idle_in_transaction_session_timeout = '30s';
ALTER ROLE backend_writer SET idle_in_transaction_session_timeout = '30s';


----- FILE: infra/postgres/init/01-databases.sh -----
#!/bin/sh
# Выполняется ОДИН РАЗ при первом создании volume.
# Только создание единой БД приложения. Роли/схемы — в db-init (bootstrap.sql), он идемпотентен.
set -e

create_db() {
  psql -v ON_ERROR_STOP=1 -U "$POSTGRES_USER" -d postgres \
    -tc "SELECT 1 FROM pg_database WHERE datname = '$1'" | grep -q 1 \
    || psql -v ON_ERROR_STOP=1 -U "$POSTGRES_USER" -d postgres -c "CREATE DATABASE \"$1\""
}

create_db "${APP_DB:-aiesec_match}"

import math
from bisect import bisect_left
from datetime import date
from typing import Dict, List, Set

from app.domain.models import Opportunity
from app.domain.profile import Profile

OPTION_WT = {"required": 1.0, "preferred": 0.5}
NEUTRAL = 0.5          # «не знаем» = середина, ни плюс ни минус


def coverage(reqs: Dict[int, str], have: Set[int]) -> float:
    """Доля требований (с весом required/preferred), которую закрывает кандидат."""
    if not reqs:
        return NEUTRAL
    total = sum(OPTION_WT.get(opt, 1.0) for opt in reqs.values())
    got = sum(OPTION_WT.get(opt, 1.0) for cid, opt in reqs.items() if cid in have)
    return got / total


def language_score(o: Opportunity, langs: Set[int]) -> float:
    preferred = {cid for cid, opt in o.languages.items() if opt == "preferred"}
    if not preferred:
        return NEUTRAL
    return len(preferred & langs) / len(preferred)


def city_score(o: Opportunity, p: Profile) -> float:
    if not p.preferred_lc_ids:
        return NEUTRAL
    return 1.0 if o.lc_id in p.preferred_lc_ids else 0.0


def duration_score(o: Opportunity, p: Profile) -> float:
    if o.duration_weeks is None or (p.min_weeks is None and p.max_weeks is None):
        return NEUTRAL
    lo = p.min_weeks if p.min_weeks is not None else 0
    hi = p.max_weeks if p.max_weeks is not None else 10**6
    d = o.duration_weeks
    dist = lo - d if d < lo else d - hi if d > hi else 0
    return max(0.0, 1 - dist / 8)          # спад до 0 на расстоянии 8 недель


def cost_score(o: Opportunity, p: Profile) -> float:
    if o.cost_total == 0:
        return 1.0
    if o.cost_total is None or p.max_budget_eur is None or o.cost_currency not in (None, "EUR"):
        return NEUTRAL
    return 1.0 if o.cost_total <= p.max_budget_eur else 0.0


def freshness_score(o: Opportunity, today: date) -> float:
    if o.date_opened is None:
        return NEUTRAL
    return math.exp(-(today - o.date_opened).days / 90)


def chance_scores(opps: List[Opportunity], a: float = 10.0) -> Dict[int, float]:
    """Шанс на принятие: сглаженная доля принятых × доля свободных мест, затем процентиль."""
    if not opps:
        return {}
    apps = sum(o.applicants_count for o in opps)
    p0 = sum(o.accepted_count for o in opps) / apps if apps else 0.0
    raw = {}
    for o in opps:
        rate = (o.accepted_count + a * p0) / (o.applicants_count + a)
        free = o.available_openings / o.openings if o.openings else 1.0
        raw[o.opportunity_id] = rate * free
    values = sorted(raw.values())
    if values[0] == values[-1]:
        return {oid: NEUTRAL for oid in raw}
    n = len(values)
    return {oid: bisect_left(values, v) / (n - 1) for oid, v in raw.items()}


def compute_features(o: Opportunity, p: Profile, langs: Set[int],
                     chance: float, today: date) -> Dict[str, float]:
    return {
        "background": coverage(o.backgrounds, p.background_ids),
        "skill": coverage(o.skills, p.skill_ids),
        "language": language_score(o, langs),
        "city": city_score(o, p),
        "duration": duration_score(o, p),
        "cost": cost_score(o, p),
        "chance": chance,
        "freshness": freshness_score(o, today),
    }

from datetime import date
from typing import List, Optional

from app.domain.models import Opportunity

def hard_filter(opps: List[Opportunity], p: Profile, filters: MatchFilters,
                min_english: str, today: Optional[date] = None) -> List[Opportunity]:
    today = today or date.today()
    langs = p.languages_for(min_english)
    out = []
    for o in opps:
        # открыта и есть места
        if not o.is_open or o.available_openings <= 0:
            continue
        # срок подачи не прошёл
        if o.applications_close_date and o.applications_close_date < today:
            continue
        # все ОБЯЗАТЕЛЬНЫЕ языки вакансии есть у кандидата
        required = {cid for cid, opt in o.languages.items() if opt == "required"}
        if not required <= langs:
            continue
        # гражданство (если кандидат его не указал, не режем)
        if o.nationalities and p.nationality_id is not None and p.nationality_id not in o.nationalities:
            continue
        # уровень учёбы (если обе стороны его знают)
        if o.study_levels and p.study_level_ids and not (o.study_levels & p.study_level_ids):
            continue
        # бюджет: платная вакансия и кандидат не готов платить или не укладывается
        if o.cost_total and o.cost_total > 0:
            if p.willing_to_pay is False:
                continue
            if (p.max_budget_eur is not None and o.cost_currency in (None, "EUR")
                    and o.cost_total > p.max_budget_eur):
                continue
        # фильтры из запроса
        if filters.country_codes and o.country_code not in filters.country_codes:
            continue
        if o.opportunity_id in filters.exclude_opportunity_ids:
            continue
        out.append(o)
    return out

from dataclasses import dataclass

from pathlib import Path
from typing import List, Optional, Set

import yaml

from app.schemas import ProfileSnapshot

CEFR = ["A1", "A2", "B1", "B2", "C1", "C2"]
ENGLISH_ID = 10018


@lru_cache
def load_taxonomy() -> dict:
    with open(Path(__file__).parent / "taxonomy.yaml", encoding="utf-8") as f:
        return yaml.safe_load(f)


@dataclass
class Profile:
    programmes: List[str]
    background_ids: Set[int]
    skill_ids: Set[int]
    language_ids: Set[int]            # все языки, кроме английского
    english_level: Optional[str]
    study_level_ids: Set[int]
    nationality_id: Optional[int]
    min_weeks: Optional[int]
    max_weeks: Optional[int]
    max_budget_eur: Optional[float]
    willing_to_pay: Optional[bool]
    preferred_lc_ids: Set[int]

    def languages_for(self, min_english: str) -> Set[int]:
        """Английский засчитываем, только если уровень не ниже порога программы."""
        ids = set(self.language_ids)
        if self.english_level and CEFR.index(self.english_level) >= CEFR.index(min_english):
            ids.add(ENGLISH_ID)
        return ids


def _sphere(name: str, tax: dict) -> dict:
    spheres = tax["spheres"]
    if name in spheres:
        return spheres[name]
    for key, val in spheres.items():          # teaching_* ловит teaching_kids и т.п.
        if key.endswith("*") and name.startswith(key[:-1]):
            return val
    return {}


def build_profile(s: ProfileSnapshot) -> Profile:
    tax = load_taxonomy()
    study = s.study or {}
    names = list(s.spheres) + list(s.activities) + ([study["field"]] if study.get("field") else [])

    bg, sk = set(), set()
    for name in names:
        m = _sphere(name, tax)
        bg.update(m.get("backgrounds", []))
        sk.update(m.get("skills", []))

    level = study.get("level")
    dur = s.duration or {}
    fin = s.financial or {}
    return Profile(
        programmes=list(s.programmes),
        background_ids=bg,
        skill_ids=sk,
        language_ids={tax["languages"][c] for c in s.other_languages if c in tax["languages"]},
        english_level=s.english_level,
        study_level_ids={tax["study_levels"][level]} if level in tax["study_levels"] else set(),
        nationality_id=tax["nationalities"].get(s.nationality) if s.nationality else None,
        min_weeks=dur.get("min_weeks"),
        max_weeks=dur.get("max_weeks"),
        max_budget_eur=fin.get("max_budget_eur"),
        willing_to_pay=fin.get("willing_to_pay"),
        preferred_lc_ids=set(s.preferred_lc_ids),
    )

@lru_cache
def load_weights() -> dict:
    with open(Path(__file__).resolve().parent.parent / "weights.yaml", encoding="utf-8") as f:
        return yaml.safe_load(f)


def label(score: float, excellent: float, good: float) -> str:
    return "excellent" if score >= excellent else "good" if score >= good else "fair"


def make_reasons(f: Dict[str, float], w: Dict[str, float]) -> List[Reason]:
    pos = sorted(((w[k] * v, k) for k, v in f.items() if w.get(k, 0) > 0 and v >= 0.7), reverse=True)
    out = [Reason(code=REASONS[k][0], polarity="positive", weight=round(c, 3), text=REASONS[k][1])
           for c, k in pos[:3]]
    if w.get("chance", 0) > 0 and f["chance"] < 0.3:
        out.append(Reason(code="chance_low", polarity="negative",
                          weight=round(w["chance"] * (1 - f["chance"]), 3),
                          text="Высокая конкуренция на эту вакансию"))
    return out


def rank_programme(opps: List[Opportunity], p: Profile, kind: str, langs: Set[int],
                   limit: int, explain: bool, excellent: float, good: float,
                   today: Optional[date] = None) -> ProgrammeGroup:
    today = today or date.today()
    w = load_weights()[kind]
    chance = chance_scores(opps)

    scored = []
    for o in opps:
        f = compute_features(o, p, langs, chance[o.opportunity_id], today)
        u = sum(w.get(k, 0) * v for k, v in f.items())
        scored.append((u, o, f))
    scored.sort(key=lambda t: (-t[0], t[1].opportunity_id))   # детерминированный порядок

    # схлопывание семей: от каждой семьи остаётся лучшая вакансия
    best, siblings = [], {}
    for u, o, f in scored:
        key = o.family_key or f"id:{o.opportunity_id}"
        if key in siblings:
            siblings[key] += 1
            continue
        siblings[key] = 0
        best.append((u, o, f, key))

    items = [
        MatchItem(
            opportunity_id=o.opportunity_id, rank=rank, score=round(u, 4),
            label=label(u, excellent, good), family_key=o.family_key,
            siblings_count=siblings[key],
            reasons=make_reasons(f, w) if explain else [],
            breakdown={k: round(v, 4) for k, v in f.items()},
        )
        for rank, (u, o, f, key) in enumerate(best[:limit], start=1)
    ]
    return ProgrammeGroup(programme_kind=kind, total_candidates=len(opps), items=items)

KIND_TO_FIELD = {"background": "backgrounds", "skill": "skills", "language": "languages"}


def _d(v) -> Optional[date]:
    if v is None or isinstance(v, date):
        return v
    return date.fromisoformat(str(v)[:10])


def build_opportunity(row: dict, reqs: List[dict]) -> Opportunity:
    """Строка вью opportunity_v1 + её строки из opportunity_requirement_v1 -> Opportunity."""
    opp = Opportunity(
        opportunity_id=int(row["opportunity_id"]),
        programme_kind=row["programme_kind"],
        title=row.get("title") or "",
        family_key=row.get("family_key"),
        country_code=row.get("country_code"),
        lc_id=row.get("lc_id"),
        is_open=bool(row.get("is_open", True)),
        openings=row.get("openings") or 0,
        available_openings=row.get("available_openings") or 0,
        applicants_count=row.get("applicants_count") or 0,
        accepted_count=row.get("accepted_count") or 0,
        date_opened=_d(row.get("date_opened")),
        applications_close_date=_d(row.get("applications_close_date")),
        duration_weeks=row.get("duration_weeks"),
        cost_total=float(row["cost_total"]) if row.get("cost_total") is not None else None,
        cost_currency=row.get("cost_currency"),
    )
    for r in reqs:
        cid = int(r["constant_id"])
        kind = r["kind"]
        if kind in KIND_TO_FIELD:
            getattr(opp, KIND_TO_FIELD[kind])[cid] = r.get("option") or "required"
        elif kind == "study_level":
            opp.study_levels.add(cid)
        elif kind == "nationality":
            opp.nationalities.add(cid)
    return opp


class FakeCatalogReader:
    """Вакансии из JSON-файла: для разработки, пока нет вью catalog."""

    def __init__(self, path: str):
        with open(path, encoding="utf-8") as f:
            data = json.load(f)
        self._all = [build_opportunity(o, o.get("requirements", [])) for o in data]

    def list_open(self, programme_kind: str) -> List[Opportunity]:
        return [o for o in self._all
                if o.programme_kind == programme_kind and o.is_open and o.available_openings > 0]


class DbCatalogReader:
    """Чтение из вью catalog.* под ролью matcher_writer (только SELECT)."""

    def __init__(self, database_url: str):
        # Драйвер должен совпадать с тем, что реально установлен в образе
        # (psycopg[binary] в matcher.txt). SQLAlchemy 2.x по умолчанию выбирает psycopg.
        if database_url.startswith("postgresql+asyncpg://"):
            database_url = database_url.replace("postgresql+asyncpg://", "postgresql+psycopg://", 1)
        elif database_url.startswith("postgresql://"):
            database_url = database_url.replace("postgresql://", "postgresql+psycopg://", 1)
        self._engine: Engine = create_engine(
            database_url, pool_size=5, max_overflow=2, pool_pre_ping=True, future=True,
        )

    def list_open(self, programme_kind: str) -> List[Opportunity]:
        with self._engine.connect() as conn:
            conn.execute(text("SET TRANSACTION READ ONLY"))
            rows = conn.execute(text("""
                SELECT * FROM catalog.opportunity_v1
                WHERE programme_kind = :k AND is_open AND available_openings > 0
            """), {"k": programme_kind}).mappings().all()
            ids = [r["opportunity_id"] for r in rows]
            reqs: Dict[int, List[dict]] = {i: [] for i in ids}
            if ids:
                for r in conn.execute(text("""
                    SELECT opportunity_id, kind, constant_id, option
                    FROM catalog.opportunity_requirement_v1
                    WHERE opportunity_id = ANY(:ids)
                """), {"ids": ids}).mappings():
                    reqs[r["opportunity_id"]].append(dict(r))
        return [build_opportunity(dict(r), reqs[r["opportunity_id"]]) for r in rows]

STRATEGIES = {"weighted_v1"}


def route_programmes(req: MatchRequest) -> Dict[str, float]:
    """Роутер v0 из ТЗ 2.2: какие программы искать и с каким приором."""
    explicit = req.filters.programme_kinds or req.profile.programmes
    if explicit:
        return {k: 1.0 for k in dict.fromkeys(explicit)}       # без дублей, порядок сохранён
    kinds = []
    spheres = req.profile.spheres
    if any(s.startswith("teaching_") for s in spheres):
        kinds.append("teacher")
    if any(not s.startswith("teaching_") for s in spheres):
        kinds.append("talent")
    if req.profile.activities:
        kinds.append("volunteer")
    return {k: round(1 / len(kinds), 3) for k in kinds}


@router.post("/match", response_model=MatchResponse)
def match(req: MatchRequest,
          reader=Depends(get_catalog_reader),
          s: Settings = Depends(get_settings)):
    started = time.perf_counter()

    if req.strategy and req.strategy not in STRATEGIES:
        raise HTTPException(422, detail={"type": "unknown-strategy"})
    prior = route_programmes(req)
    if not prior:
        raise HTTPException(422, detail={"type": "profile-incomplete", "missing": ["programmes"]})

    p = build_profile(req.profile)
    groups = []
    for kind in prior:
        min_en = s.min_english.get(kind, "B1")
        try:
            opps = reader.list_open(kind)
        except Exception as e:                       # БД недоступна
            log.exception("catalog read failed kind=%s", kind)
            raise HTTPException(503, detail={"type": "catalog-unavailable"}) from e
        cands = hard_filter(opps, p, req.filters, min_en)
        groups.append(rank_programme(cands, p, kind, p.languages_for(min_en),
                                     req.limit_per_programme, req.explain,
                                     s.label_excellent, s.label_good))

    took = int((time.perf_counter() - started) * 1000)
    log.info("match profile_id=%s kinds=%s took_ms=%s", req.profile_id, list(prior), took)
    return MatchResponse(request_id=req.request_id, profile_id=req.profile_id,
                         programme_prior=prior, took_ms=took, groups=groups)

@router.put("/admin/token", tags=["admin"])
def put_token(body: TokenIn, bg: BackgroundTasks):
    was_failing = client.auth_ok is False
    verified = None

    if body.refresh_token:
        if not client.refresher:
            raise HTTPException(422, "Автообновление не настроено: задайте AIESEC_CLIENT_ID и AIESEC_CLIENT_SECRET")
        try:
            client.refresh_with(_clean(body.refresh_token))
        except AuthError as e:
            raise HTTPException(422, f"{e}")
        except AiesecError as e:
            raise HTTPException(502, f"Не удалось обменять refresh_token: {e}")
        verified = True
    else:
        new = _clean(body.token)
        if len(new) < 10:
            raise HTTPException(422, "Токен слишком короткий после очистки")
        if body.verify:
            try:
                ok, err = client.check_token(new)
            except AiesecError as e:
                raise HTTPException(
                    502,
                    f"Не удалось проверить токен: {e}. Старый токен не менялся. "
                    f"Повторите позже или передайте verify=false.",
                )
            if not ok:
                raise HTTPException(422, f"AIESEC отклонил новый токен, оставлен прежний. Ответ: {err}")
            verified = True
        client.set_token(new, source="api", verified=verified)
        persist_tokens(client)

    triggered = None
    if body.run_sync and (was_failing or not service.has_bootstrapped()):
        kind = "incremental" if service.has_bootstrapped() else "full"
        run_id = service.try_start(kind)
        if run_id is not None:
            bg.add_task(service.execute, run_id, kind)
            triggered = {"run_id": run_id, "kind": kind}
    return {**_token_status(), "verified": verified, "sync": triggered}

def to_dict(row: Opportunity, full: bool = False) -> dict:

def _actual_clause():
    return (Opportunity.is_open.is_(True)) & (
        Opportunity.applications_close_date.is_(None)
        | (Opportunity.applications_close_date >= date.today())
    )


@router.get("/opportunities", tags=["opportunities"])
def list_opportunities(
    country: str | None = Query(None, description="Название страны из targets.json"),
    city: str | None = Query(None, description="Город Турции из targets.json"),
    home_mc_id: int | None = None,
    home_lc_id: int | None = None,
    programme_id: int | None = None,
    q: str | None = Query(None, description="Поиск по названию"),
    only_open: bool = True,
    closing_before: date | None = None,
    new_since: datetime | None = Query(None, description="Впервые увидели после (UTC)"),
    updated_since: datetime | None = Query(
        None,
        description="Изменились/закрылись после (UTC). Используйте с only_open=false",
    ),
    sort: Literal["first_seen_at", "applications_close_date", "date_opened", "updated_at",
                  "remote_updated_at"] = "first_seen_at",
    desc: bool = True,
    limit: int = Query(50, ge=1, le=500),
    offset: int = Query(0, ge=0),
    db: Session = Depends(get_db),
):
    conds = []
    if only_open:
        conds.append(_actual_clause())
    if country:
        mc = next((v for k, v in targets.countries.items() if k.lower() == country.lower()), None)
        if mc is None:
            raise HTTPException(404, f"Страна '{country}' не в targets.json")
        conds.append(Opportunity.home_mc_id == mc)
    if city:
        lc = next((v for k, v in targets.turkey_cities.items() if k.lower() == city.lower()), None)
        if lc is None:
            raise HTTPException(404, f"Город '{city}' не в targets.json")
        conds.append(or_(Opportunity.home_lc_id == lc, Opportunity.host_lc_id == lc))
    if home_mc_id is not None:
        conds.append(Opportunity.home_mc_id == home_mc_id)
    if home_lc_id is not None:
        conds.append(Opportunity.home_lc_id == home_lc_id)
    if programme_id is not None:
        conds.append(Opportunity.programme_id == programme_id)
    if q:
        conds.append(Opportunity.title.ilike(f"%{q}%"))
    if closing_before:
        conds.append(Opportunity.applications_close_date <= closing_before)
    if new_since:
        conds.append(Opportunity.first_seen_at >= new_since)
    if updated_since:

"""Обмен refresh_token -> access_token (OAuth2, auth.aiesec.org)."""
from __future__ import annotations

import logging
from dataclasses import dataclass
from datetime import datetime, timedelta, timezone

import requests

from .errors import AiesecError, AuthError

log = logging.getLogger("aiesec.auth")


@dataclass
class TokenSet:
    access_token: str
    refresh_token: str | None
    expires_at: datetime | None
    scope: str | None = None


def utcnow() -> datetime:
    return datetime.now(timezone.utc).replace(tzinfo=None)


class OAuthRefresher:
    def __init__(self, url: str, client_id: str, client_secret: str, origin: str | None = None,
                 timeout: int = 30, session: requests.Session | None = None):
        self.url = url
        self.client_id = client_id
        self.client_secret = client_secret
        self.origin = origin
        self.timeout = timeout
        self._http = session or requests.Session()

    def refresh(self, refresh_token: str) -> TokenSet:
        headers = {"Accept": "application/json"}
        if self.origin:
            headers["Origin"] = self.origin
        try:
            r = self._http.post(
                self.url,
                auth=(self.client_id, self.client_secret),
                data={"grant_type": "refresh_token", "refresh_token": refresh_token},
                headers=headers,
                timeout=self.timeout,
            )
        except requests.RequestException as e:
            raise AiesecError(f"сервер авторизации AIESEC недоступен: {e}") from e

        if r.status_code in (400, 401, 403):
            raise AuthError(f"refresh_token отклонён ({r.status_code}): {r.text[:300]}. "
                            f"Нужен новый refresh_token: PUT /admin/token")
        if r.status_code != 200:
            raise AiesecError(f"сервер авторизации ответил HTTP {r.status_code}: {r.text[:200]}")
        try:
            j = r.json()
            access = j["access_token"]
        except (ValueError, KeyError) as e:
            raise AiesecError(f"неожиданный ответ сервера авторизации: {r.text[:200]}") from e

        return TokenSet(access_token=access, refresh_token=j.get("refresh_token"),
                        expires_at=self._expires_at(j), scope=j.get("scope"))

    @staticmethod
    def _expires_at(j: dict) -> datetime | None:
        try:
            ttl = int(j["expires_in"])
        except (KeyError, TypeError, ValueError):
            return None
        now = utcnow()
        local = now + timedelta(seconds=ttl)
        try:
            by_server = datetime.fromtimestamp(int(j["created_at"]) + ttl, timezone.utc).replace(tzinfo=None)
        except (KeyError, TypeError, ValueError, OverflowError):
            return local
        return by_server if abs((by_server - local).total_seconds()) < 300 else local

TYPE_FRAG = "fragment T on __Type{kind name ofType{kind name ofType{kind name ofType{kind name}}}}"


def type_str(t: dict) -> str:
    if t["kind"] == "NON_NULL":
        return type_str(t["ofType"]) + "!"
    if t["kind"] == "LIST":
        return "[" + type_str(t["ofType"]) + "]"
    return t["name"]


def unwrap(t: dict) -> dict:
    while t.get("ofType"):
        t = t["ofType"]
    return t


def norm(s: Any) -> str:
    s = unicodedata.normalize("NFKD", str(s or "")).encode("ascii", "ignore").decode().lower()
    return " ".join(s.split())


@dataclass
class SearchSchema:
    field: str  # searchOpportunities | opportunities
    args: dict[str, str]
    item_type: str
    filters: dict[str, dict]  # имя фильтра -> inputField


class AiesecClient:
    def __init__(self, token: str, url: str, per_page: int = 200, max_pages: int = 200,
                 delay: float = 0.7, timeout: int = 60, max_retries: int = 4,
                 exclude_fields: list[str] | None = None,
                 refresher: OAuthRefresher | None = None, refresh_token: str | None = None,
                 refresh_margin: int = 600):
        self.token = (token or "").strip()
        self.url = url
        self.per_page = per_page
        self.max_pages = max_pages
        self.delay = delay
        self.timeout = timeout
        self.max_retries = max_retries
        self.broken_fields: set[str] = set(exclude_fields or [])

        # состояние авторизации: None — ещё не проверяли, True/False — по результату последнего запроса
        self.token_source = "env"
        self.auth_ok: bool | None = None
        self.auth_error: str | None = None
        self.auth_checked_at: datetime | None = None

        # автообновление: refresh_token -> access_token
        self.refresher = refresher
        self.refresh_token: str | None = (refresh_token or "").strip() or None
        self.expires_at: datetime | None = None  # naive UTC; None — срок жизни неизвестен
        self.refresh_margin = refresh_margin
        self.on_tokens: Callable[["AiesecClient"], None] | None = None  # сохранить токены (вызывается до их использования)
        self._refresh_lock = threading.RLock()

        self._http = requests.Session()
        self._throttle_lock = threading.Lock()
        self._last_call = 0.0
        self._types: dict[str, dict] = {}
        self._qf: dict[str, dict] | None = None
        self._schema: SearchSchema | None = None
        self._select_cache: dict[tuple, str] = {}

    # ---------- транспорт ----------
    def _throttle(self) -> None:
        with self._throttle_lock:
            wait = self._last_call + self.delay - time.monotonic()
            if wait > 0:
                time.sleep(wait)
            self._last_call = time.monotonic()

    def gql(self, query: str, variables: dict | None = None) -> dict:
        last_err: Exception | None = None
        refreshed_after_401 = False
        self.ensure_fresh()  # проактивно: до истечения access_token
        for attempt in range(self.max_retries + 1):
            self._throttle()
            used_token = self.token
            try:
                r = self._http.post(
                    self.url,
                    params={"access_token": used_token},
                    headers={"Authorization": used_token, "Content-Type": "application/json"},
                    json={"query": query, "variables": variables or {}},
                    timeout=self.timeout,
                )
            except requests.RequestException as e:
                last_err = e
                self._backoff(attempt)
                continue
            if r.status_code == 401 and not refreshed_after_401 and self.can_refresh:
                # реактивно: токен умер раньше срока (отозван, рассинхрон часов) — один раз освежаем и повторяем
                refreshed_after_401 = True
                log.warning("401 от AIESEC — пробую обновить токен через refresh_token")
                self.refresh_after_401(used_token)  # AuthError, если и refresh_token мёртв
                continue
            if r.status_code in (401, 403):
                msg = f"{r.status_code}: токен не принят. {r.text[:300]}"
                self._mark_auth(False, msg)
                raise AuthError(msg)
            if r.status_code == 429 or r.status_code >= 500:
                last_err = AiesecError(f"HTTP {r.status_code}")
                retry_after = None
                try:
                    retry_after = float(r.headers.get("Retry-After", ""))
                except ValueError:
                    pass
                log.warning("AIESEC вернул %s, попытка %s/%s", r.status_code, attempt + 1, self.max_retries + 1)
                self._backoff(attempt, retry_after)
                continue
            r.raise_for_status()
            self._mark_auth(True)
            data = r.json()
            if data.get("errors"):
                raise GQLError(data["errors"])
            return data["data"]
        raise AiesecError(f"запрос не удался после ретраев: {last_err}")

    def _backoff(self, attempt: int, retry_after: float | None = None) -> None:
        time.sleep(retry_after if retry_after else min(2 ** attempt * 2, 60))

    # ---------- токен ----------
    def _mark_auth(self, ok: bool, error: str | None = None) -> None:
        self.auth_ok = ok
        self.auth_error = None if ok else error
        self.auth_checked_at = datetime.now(timezone.utc).replace(tzinfo=None)

    @property
    def can_refresh(self) -> bool:
        return bool(self.refresher and self.refresh_token)

    def seconds_left(self) -> float | None:
        if not self.expires_at:
            return None
        return (self.expires_at - utcnow()).total_seconds()

    def set_token(self, token: str, source: str = "api", verified: bool | None = None,
                  refresh_token: str | None = None, expires_at: datetime | None = None) -> None:
        """Подмена токена на лету. Кэши схемы не трогаем: схема от токена не зависит.
        refresh_token=None — оставить прежний; expires_at=None — срок неизвестен."""
        self.token = token.strip()
        if refresh_token is not None:
            self.refresh_token = refresh_token.strip() or None
        self.expires_at = expires_at
        self.token_source = source
        self.auth_ok = verified
        self.auth_error = None
        self.auth_checked_at = None

    def _needs_refresh(self, refresh_if_unknown: bool) -> bool:
        left = self.seconds_left()
        if not self.token:
            return True
        if left is None:
            return refresh_if_unknown
        return left < self.refresh_margin

    def ensure_fresh(self, refresh_if_unknown: bool = False) -> None:
        """Освежить токен, если он скоро истечёт. Временные сбои глотаем (токен мог ещё быть живым),
        AuthError (мёртвый refresh_token) пробрасываем."""
        if not self.can_refresh:
            return
        with self._refresh_lock:
            if self._needs_refresh(refresh_if_unknown):
                self._do_refresh(proactive=True)

    def refresh_after_401(self, stale_token: str) -> None:
        """Освежить токен после 401. Если другой поток уже успел — второй раз refresh_token НЕ тратим
        (он одноразовый, повторное использование сломало бы цепочку)."""
        with self._refresh_lock:
            if self.token != stale_token:
                return
            self._do_refresh(proactive=False)

    def refresh_with(self, refresh_token: str):
        """Обмен произвольного refresh_token (ручной ввод через API). Применяет и сохраняет результат."""
        if not self.refresher:
            raise AuthError("не настроены AIESEC_CLIENT_ID / AIESEC_CLIENT_SECRET")
        with self._refresh_lock:
            ts = self.refresher.refresh(refresh_token)
            self._apply(ts, fallback_refresh=refresh_token)
            return ts

    def _do_refresh(self, proactive: bool) -> None:
        try:
            ts = self.refresher.refresh(self.refresh_token)
        except AuthError as e:
            self._mark_auth(False, str(e))
            raise
        except AiesecError as e:
            if proactive:
                log.warning("не удалось обновить токен заранее (%s), продолжаю со старым", e)
                return
            raise
        self._apply(ts, fallback_refresh=self.refresh_token)

    def _apply(self, ts, fallback_refresh: str | None) -> None:
        self.token = ts.access_token
        self.refresh_token = ts.refresh_token or fallback_refresh
        self.expires_at = ts.expires_at
        self.token_source = "refresh"
        self._mark_auth(True)
        log.info("токен AIESEC обновлён, действует ещё ~%s с", int(self.seconds_left() or 0))
        if self.on_tokens:
            try:  # потеря новой пары = потеря цепочки refresh_token после рестарта, поэтому шумим громко
                self.on_tokens(self)
            except Exception:  # noqa: BLE001
                log.exception("КРИТИЧНО: не удалось сохранить новые токены, после рестарта потребуется новый refresh_token")

    def check_token(self, token: str | None = None) -> tuple[bool, str | None]:
        """Один дешёвый запрос. (True, None) / (False, ответ AIESEC). AiesecError — если AIESEC недоступен.
        Токен-кандидат не применяется; если проверяем текущий — обновляем состояние авторизации."""
        candidate = (token or self.token).strip()
        self._throttle()
        try:
            r = self._http.post(self.url, params={"access_token": candidate},
                                headers={"Authorization": candidate, "Content-Type": "application/json"},
                                json={"query": "{ __typename }"}, timeout=self.timeout)
        except requests.RequestException as e:
            raise AiesecError(f"AIESEC недоступен: {e}") from e
        if r.status_code == 429 or r.status_code >= 500:
            raise AiesecError(f"AIESEC ответил HTTP {r.status_code}, проверить токен не удалось")
        ok = r.status_code not in (401, 403)
        if candidate == self.token:
            self._mark_auth(ok, None if ok else f"{r.status_code}: {r.text[:300]}")
        return ok, (None if ok else r.text[:300])

    # ---------- интроспекция ----------
    def get_type(self, name: str) -> dict:
        if name not in self._types:
            q = ("query($n:String!){__type(name:$n){name kind "
                 "fields{name args{name type{...T}} type{...T}} "
                 "inputFields{name type{...T}} enumValues{name}}} " + TYPE_FRAG)
            self._types[name] = self.gql(q, {"n": name})["__type"]
        return self._types[name]

    def query_fields(self) -> dict[str, dict]:
        if self._qf is None:
            q = "{__schema{queryType{fields{name args{name type{...T}} type{...T}}}}} " + TYPE_FRAG
            self._qf = {f["name"]: f for f in self.gql(q)["__schema"]["queryType"]["fields"]}
        return self._qf

    @staticmethod
    def arg_types(field: dict) -> dict[str, str]:
        return {a["name"]: type_str(a["type"]) for a in field["args"]}

    def schema(self) -> SearchSchema:
        if self._schema:
            return self._schema
        qf = self.query_fields()
        fname = next((n for n in SEARCH_FIELDS if n in qf), None)
        if not fname:
            raise AiesecError("В схеме нет searchOpportunities/opportunities")
        field = qf[fname]
        args = self.arg_types(field)
        list_t = self.get_type(unwrap(field["type"])["name"])
        data = next((f for f in list_t["fields"] if f["name"] == "data"), None)
        if not data:
            raise AiesecError("В типе списка нет поля data")
        filters_t = self.get_type(args["filters"].rstrip("!"))
        self._schema = SearchSchema(
            field=fname, args=args, item_type=unwrap(data["type"])["name"],
            filters={f["name"]: f for f in filters_t["inputFields"]},
        )
        return self._schema

    def supports_range_filter(self, name: str) -> bool:
        """Есть ли фильтр `name` вида {from, to}."""
        f = self.schema().filters.get(name)
        if not f:
            return False
        base = unwrap(f["type"])
        if base["kind"] != "INPUT_OBJECT":
            return False
        return "from" in {x["name"] for x in self.get_type(base["name"])["inputFields"] or []}

    def _scalar_names(self, type_name: str) -> set[str]:
        return {f["name"] for f in self.get_type(type_name)["fields"]
                if unwrap(f["type"])["kind"] in ("SCALAR", "ENUM")}

    def _auto_select(self, type_name: str) -> str:
        parts = []
        for f in self.get_type(type_name)["fields"]:
            name = f["name"]
            if name in self.broken_fields:
                continue
            base = unwrap(f["type"])
            if base["kind"] == "ENUM":
                parts.append(name)
            elif base["kind"] == "SCALAR":
                if base["name"] == "JSON" and name not in JSON_KEEP:
                    continue
                parts.append(name)
            elif base["kind"] == "OBJECT" and name in NESTED_PREFERRED:
                sub = [sf["name"] for sf in self.get_type(base["name"])["fields"]
                       if sf["name"] in NESTED_PREFERRED[name] and sf["name"] not in self.broken_fields
                       and unwrap(sf["type"])["kind"] in ("SCALAR", "ENUM")]
                parts.append(f"{name}{{{' '.join(sub or ['id'])}}}")
            elif base["kind"] == "OBJECT" and name in LIST_OBJECT_FIELDS:
                sub = [sf["name"] for sf in self.get_type(base["name"])["fields"]
                       if unwrap(sf["type"])["kind"] in ("SCALAR", "ENUM")
                       and unwrap(sf["type"])["name"] != "JSON" and sf["name"] not in self.broken_fields]
                if sub:
                    parts.append(f"{name}{{{' '.join(sub)}}}")
        return " ".join(parts)

    def select(self, mode: str, type_name: str | None = None) -> str:
        type_name = type_name or self.schema().item_type
        key = (type_name, mode, frozenset(self.broken_fields))
        if key not in self._select_cache:
            if mode == "slim":
                have = self._scalar_names(type_name)
                self._select_cache[key] = " ".join(n for n in SLIM_FIELDS
                                                   if n in have and n not in self.broken_fields)
            else:
                self._select_cache[key] = self._auto_select(type_name)
        return self._select_cache[key]

    # ---------- запросы с авто-исключением «сломанных» полей ----------
    def _run(self, make_query: Callable[[], str], variables: dict, skip: set[str],
             extract: Callable[[dict], Any]) -> Any:
        for _ in range(20):
            try:
                return extract(self.gql(make_query(), variables))
            except GQLError as e:
                new = {err["path"][-1] for err in e.errors
                       if err.get("path") and isinstance(err["path"][-1], str)} - self.broken_fields - skip
                if not new:
                    raise
                self.broken_fields |= new
                self._select_cache.clear()
                log.warning("сервер не умеет поля %s — исключаю и повторяю", sorted(new))
        raise AiesecError("слишком много сломанных полей")

    def search_page(self, filters: dict, page: int, mode: str = "full") -> list[dict]:
        s = self.schema()

        def make_query() -> str:
            a = s.args
            return (f"query($page:{a.get('page', 'Int')},$per_page:{a.get('per_page', 'Int')},"
                    f"$filters:{a['filters']}){{ {s.field}(page:$page, per_page:$per_page, filters:$filters)"
                    f"{{ data {{ {self.select(mode)} }} }} }}")

        return self._run(make_query, {"page": page, "per_page": self.per_page, "filters": filters},
                         {s.field, "data"}, lambda d: d[s.field]["data"] or [])

    def iter_pages(self, filters: dict, mode: str = "full") -> Iterator[list[dict]]:
        """Отдаёт страницы. Если упёрлись в max_pages — ошибка (выдача неполная, закрывать по ней нельзя)."""
        for page in range(1, self.max_pages + 1):
            rows = self.search_page(filters, page, mode)
            log.info("стр. %s: %s (%s)", page, len(rows), mode)
            if rows:
                yield rows
            if len(rows) < self.per_page:
                return
        raise AiesecError(f"достигнут max_pages={self.max_pages}, выдача неполная")

    def get_opportunity(self, opportunity_id: int) -> dict | None:
        f = self.query_fields().get("opportunity")
        if not f:
            raise NotSupported("в схеме нет запроса opportunity(id)")
        id_type = self.arg_types(f).get("id", "ID!")
        type_name = unwrap(f["type"])["name"]

        def make_query() -> str:
            return f"query($id:{id_type}){{ opportunity(id:$id){{ {self.select('full', type_name)} }} }}"

        try:
            return self._run(make_query, {"id": opportunity_id}, {"opportunity"}, lambda d: d["opportunity"])
        except GQLError as e:
            if "not found" in str(e).lower():
                return None
            raise

    # ---------- офисы ----------
    def search_offices(self, q: str, limit: int = 50) -> list[dict]:
        query = ("query($q:String,$pp:Int){ committees(q:$q, per_page:$pp)"
                 "{ data { id name full_name tag country_code parent{ id name } } } }")
        return self.gql(query, {"q": q, "pp": limit})["committees"]["data"] or []

from datetime import date, datetime
from typing import Any

_MISSING = object()

SCALARS = {
    "title": "title",
    "status": "status",
    "current_status": "current_status",
    "location": "location",
    "openings": "openings",
    "available_openings": "available_openings",
    "applicants_count": "applicants_count",
    "accepted_count": "accepted_count",
    "duration": "duration",
    "experience_type": "experience_type",
    "remote_updated_at": "updated_at",
    "date_opened": "date_opened",
    "applications_close_date": "applications_close_date",
    "earliest_start_date": "earliest_start_date",
    "latest_end_date": "latest_end_date",
    "skills": "skills",
    "backgrounds": "backgrounds",
    "languages": "languages",
    "study_levels": "study_levels",
    "nationalities": "nationalities",
    "skills_developments": "skills_developments",
}

NESTED = {
    "programme_id": [("programme", "id")],
    "programme_name": [("programme", "short_name_display"), ("programme", "short_name"), ("programme", "name")],
    "home_mc_id": [("home_mc", "id")],
    "home_mc_name": [("home_mc", "name"), ("home_mc", "full_name")],
    "home_lc_id": [("home_lc", "id")],
    "home_lc_name": [("home_lc", "name"), ("home_lc", "full_name")],
    "host_lc_id": [("host_lc", "id")],
    "host_lc_name": [("host_lc", "name"), ("host_lc", "full_name")],
    "organisation_id": [("organisation", "id")],
    "organisation_name": [("organisation", "name")],
    "cost_total": [("opportunity_cost", "total")],
    "cost_currency": [("opportunity_cost", "currency")],
}

INT_COLS = {"programme_id", "home_mc_id", "home_lc_id", "host_lc_id", "organisation_id",
            "openings", "available_openings", "applicants_count", "accepted_count", "duration"}
DATE_COLS = {"date_opened", "applications_close_date", "earliest_start_date", "latest_end_date"}
DATETIME_COLS = {"remote_updated_at"}


def to_int(v: Any) -> int | None:
    try:
        return int(v)
    except (TypeError, ValueError):
        return None


def to_float(v: Any) -> float | None:
    try:
        return float(v)
    except (TypeError, ValueError):
        return None


def to_date(v: Any) -> date | None:
    if not v:
        return None
    try:
        return date.fromisoformat(str(v)[:10])
    except ValueError:
        return None


def to_datetime(v: Any) -> datetime | None:
    """ISO-8601 из AIESEC (обычно с 'Z') -> naive UTC datetime."""
    if not v:
        return None
    try:
        s = str(v).replace("Z", "+00:00")
        return datetime.fromisoformat(s).replace(tzinfo=None)
    except ValueError:
        return None


def _dig(raw: dict, path: tuple[str, ...]) -> Any:
    cur: Any = raw
    for i, k in enumerate(path):
        if not isinstance(cur, dict) or k not in cur:
            return None if (i > 0 and cur is None) else _MISSING
        cur = cur[k]
    return cur


def normalize(raw: dict, full: bool = True) -> dict:
    out: dict[str, Any] = {"opportunity_id": int(raw["id"])}
    for col, key in SCALARS.items():
        if key in raw:
            out[col] = raw[key]
    for col, paths in NESTED.items():
        vals = [_dig(raw, p) for p in paths]
        present = [v for v in vals if v is not _MISSING]
        if present:
            out[col] = next((v for v in present if v not in (None, "")), None)
    for col in INT_COLS & out.keys():
        out[col] = to_int(out[col])
    for col in DATE_COLS & out.keys():
        out[col] = to_date(out[col])
    for col in DATETIME_COLS & out.keys():
        out[col] = to_datetime(out[col])
    if "cost_total" in out:
        out["cost_total"] = to_float(out["cost_total"])
    if full:
        out["raw"] = raw
    return out

def gap_fill(
    client: AiesecClient, ingestor: Ingestor, ids: set[int],
    stats: Counter, full_run,
) -> None:
    if len(ids) <= ingestor.settings.gap_fill_max:
        try:
            with ingestor.Session() as s:
                now = utcnow()
                for oid in ids:
                    raw = client.get_opportunity(oid)
                    if raw:
                        stats[ingestor.upsert_one(s, raw, "gap_fill", True, now.date(), now)] += 1
                s.commit()
            stats["gap_fill"] = "by_id"
            return
        except NotSupported:
            log.warning("opportunity(id) недоступен — догружаю полным проходом")
    stats["gap_fill"] = "full_scan"
    full_run(stats)

"""Upsert + actuality + close_missing. Единственный класс, который пишет в raw.opportunities."""
from __future__ import annotations

import logging
from collections import Counter
from datetime import date, datetime

from sqlalchemy import select, update
from sqlalchemy.orm import Session, sessionmaker

from ..client import AiesecError
from ..config import Settings, Targets
from ..models import Opportunity, utcnow
from ..normalize import normalize

log = logging.getLogger("aiesec.sync.ingest")


class Ingestor:
    def __init__(self, settings: Settings, targets: Targets, session_factory: sessionmaker):
        self.settings = settings
        self.targets = targets
        self.Session = session_factory

    def actuality(self, d: dict, today: date, check_target: bool) -> tuple[bool, str | None]:
        t = self.targets
        if check_target:
            in_country = d.get("home_mc_id") in t.mc_ids
            in_city = d.get("home_lc_id") in t.lc_ids or d.get("host_lc_id") in t.lc_ids
            if not (in_country or in_city):
                return False, "out_of_target"
        status = d.get("status")
        if status is not None and "open" not in str(status).lower():
            return False, "status_not_open"
        close = d.get("applications_close_date")
        if close and close < today:
            return False, "deadline_passed"
        slots = d.get("available_openings")
        if t.only_with_openings and isinstance(slots, (int, float)) and slots <= 0:
            return False, "no_openings"
        return True, None

    def upsert_one(
        self, s: Session, raw: dict, source: str, full: bool, today: date, now: datetime,
    ) -> str:
        d = normalize(raw, full=full)
        ok, reason = self.actuality(d, today, check_target=full)
        row = s.get(Opportunity, d["opportunity_id"])

        if row is None:
            if not ok:
                return f"skipped_{reason}"
            s.add(Opportunity(**d, is_open=True, source_query=source,
                              first_seen_at=now, last_seen_at=now, updated_at=now))
            return "inserted"

        was_open = row.is_open
        for k, v in d.items():
            setattr(row, k, v)
        row.last_seen_at = now
        row.updated_at = now
        if ok:
            row.is_open, row.close_reason, row.closed_at = True, None, None
            return "updated" if was_open else "reopened"
        if was_open:
            row.is_open, row.close_reason, row.closed_at = False, reason, now
            return "closed"
        row.close_reason = row.close_reason or reason
        return "still_closed"

    def ingest_page(
        self, s: Session, rows: list[dict], source: str,
        stats: Counter, seen: set[int], full: bool,
    ) -> None:
        today, now = date.today(), utcnow()
        for raw in rows:
            seen.add(int(raw["id"]))
            stats[self.upsert_one(s, raw, source, full, today, now)] += 1
        s.commit()

    def close_missing(
        self, s: Session, seen: set[int], stats: Counter, reason: str = "not_in_open_list",
    ) -> None:
        open_ids = set(s.scalars(select(Opportunity.opportunity_id).where(Opportunity.is_open.is_(True))))
        missing = open_ids - seen
        if not missing:
            return
        if len(open_ids) >= 20 and len(missing) / len(open_ids) > self.settings.max_close_ratio:
            raise AiesecError(
                f"Предохранитель: к закрытию {len(missing)} из {len(open_ids)} открытых "
                f"(> {self.settings.max_close_ratio:.0%}). Похоже на неполную выдачу, ничего не закрываю."
            )
        now = utcnow()
        ids = list(missing)
        for i in range(0, len(ids), 500):
            s.execute(
                update(Opportunity)
                .where(Opportunity.opportunity_id.in_(ids[i:i + 500]))
                .values(is_open=False, close_reason=reason, closed_at=now, updated_at=now)
            )
        s.commit()
        stats["closed"] += len(missing)

def build_queries(client: AiesecClient, targets: Targets, extra: dict | None = None) -> list[tuple[str, dict]]:
    known = client.schema().filters
    out = []
    for name, geo in targets.geo_queries():
        bad_geo = [k for k in geo if k not in known]
        if bad_geo:
            raise AiesecError(f"Схема не поддерживает гео-фильтры {bad_geo}")
        filters = {**targets.base_filters, **(extra or {}), **geo}
        for k in [k for k in filters if k not in known]:
            log.warning("фильтр '%s' не поддерживается схемой — отбрасываю", k)
            filters.pop(k)
        out.append((name, filters))
    if not out:
        raise AiesecError("В targets.json нет ни стран, ни городов")
    return out

"""Публичный фасад SyncService. Сохраняет прежний API для API-слоя и тестов."""
from __future__ import annotations

import logging
import threading

from sqlalchemy import update

from ..client import AiesecClient, AuthError

class SyncService:
    def __init__(self, client: AiesecClient, settings: Settings, targets: Targets,
                 session_factory: sessionmaker):
        self.client = client
        self.settings = settings
        self.targets = targets
        self.Session = session_factory
        self.state = StateStore(session_factory)
        self.ingestor = Ingestor(settings, targets, session_factory)
        full = FullStrategy(client, settings, targets, self.ingestor, self.state)
        incremental = IncrementalStrategy(client, settings, targets, self.ingestor, self.state, full)
        reconcile = ReconcileStrategy(client, settings, targets, self.ingestor, self.state, full)
        self._strategies = {"full": full, "incremental": incremental, "reconcile": reconcile}
        self._lock = threading.Lock()

    # ---------- запуск ----------
    def is_running(self) -> bool:
        return self._lock.locked()

    def try_start(self, kind: str) -> int | None:
        if not self._lock.acquire(blocking=False):
            return None
        try:
            with self.Session() as s:
                run = SyncRun(kind=kind)
                s.add(run)
                s.commit()
                return run.run_id
        except Exception:
            self._lock.release()
            raise

    def execute(self, run_id: int, kind: str) -> None:
        stats: Counter = Counter()
        status, error = "ok", None
        try:
            self._strategies[kind].run(stats)
        except AuthError as e:
            log.error("AIESEC отклонил токен: %s", e)
            status, error = "auth_failed", f"AuthError: {e}"[:2000]
        except Exception as e:  # noqa: BLE001
            log.exception("sync %s упал", kind)
            status, error = "failed", f"{type(e).__name__}: {e}"[:2000]
        finally:
            with self.Session() as s:
                run = s.get(SyncRun, run_id)
                run.status, run.error, run.stats, run.finished_at = status, error, dict(stats), utcnow()
                s.commit()
            self._lock.release()

    def run_now(self, kind: str) -> None:
        if getattr(self.client, "auth_ok", None) is False:
            log.warning("sync %s пропущен: токен AIESEC отклонён. Обновите его: PUT /admin/token", kind)
            return
        run_id = self.try_start(kind)
        if run_id is None:
            log.info("sync %s пропущен: уже выполняется другой", kind)
            return
        self.execute(run_id, kind)

    def run_sync_for_test(self, kind: str) -> None:
        """Публичный запуск стратегии без блокировки. Только для тестов и ручного дерганья."""
        self._strategies[kind].run(Counter())

    # ---------- состояние ----------
    def get_state(self, key: str) -> str | None:
        return self.state.get(key)

    def set_state(self, key: str, value: str) -> None:
        self.state.set(key, value)

    def set_states(self, items: dict[str, str]) -> None:
        self.state.set_many(items)

    def has_bootstrapped(self) -> bool:
        return self.get_state("last_full_at") is not None

    # ---------- вне блокировки ----------
    def expire_by_deadline(self) -> int:
        now = utcnow()
        with self.Session() as s:
            res = s.execute(
                update(Opportunity)
                .where(
                    Opportunity.is_open.is_(True),
                    Opportunity.applications_close_date.is_not(None),
                    Opportunity.applications_close_date < date.today(),
                )
                .values(is_open=False, close_reason="deadline_passed", closed_at=now, updated_at=now)
            )
            s.commit()
            if res.rowcount:
                log.info("закрыто по дедлайну: %s", res.rowcount)
            return res.rowcount or 0

    def refresh_one(self, opportunity_id: int) -> dict:
        raw = self.client.get_opportunity(opportunity_id)  # NotSupported пробросится наверх
        if raw is None:
            return {"result": "not_found_remote"}
        with self.Session() as s:
            result = self.ingestor.upsert_one(s, raw, "refresh", True, date.today(), utcnow())
            s.commit()
        return {"result": result}

"""Три стратегии синка: full, incremental, reconcile."""
from __future__ import annotations

import logging
from collections import Counter
from datetime import date, datetime, timedelta

from sqlalchemy import select

from ..client import AiesecClient, AiesecError
from ..config import Settings, Targets
from ..models import Opportunity, utcnow
from ..normalize import normalize
from .gapfill import gap_fill
from .ingest import Ingestor
from .queries import build_queries
from .state import StateStore

log = logging.getLogger("aiesec.sync.strategies")


class FullStrategy:
    def __init__(self, client: AiesecClient, settings: Settings, targets: Targets,
                 ingestor: Ingestor, state: StateStore):
        self.client, self.settings, self.targets = client, settings, targets
        self.ingestor, self.state = ingestor, state

    def run(self, stats: Counter) -> None:
        started = utcnow()
        seen: set[int] = set()
        with self.ingestor.Session() as s:
            for name, filters in build_queries(self.client, self.targets):
                log.info("full → %s: %s", name, filters)
                for rows in self.client.iter_pages(filters, "full"):
                    self.ingestor.ingest_page(s, rows, name, stats, seen, full=True)
            stats["fetched"] = len(seen)
            self.ingestor.close_missing(s, seen, stats)
        self.state.set("last_full_at", started.isoformat())
        self.state.set("last_incremental_at", started.isoformat())


class IncrementalStrategy:
    def __init__(self, client: AiesecClient, settings: Settings, targets: Targets,
                 ingestor: Ingestor, state: StateStore, full: FullStrategy):
        self.client, self.settings, self.targets = client, settings, targets
        self.ingestor, self.state, self.full = ingestor, state, full

    def run(self, stats: Counter) -> None:
        started = utcnow()
        df = self.settings.incremental_date_filter
        anchor = self.state.get("last_incremental_at") or self.state.get("last_full_at")
        if not anchor:
            stats["fallback"] = "full (первого полного прохода ещё не было)"
            return self.full.run(stats)
        if not self.client.supports_range_filter(df):
            log.warning("фильтр '%s' {from,to} не поддерживается — делаю full", df)
            stats["fallback"] = f"full (нет фильтра {df})"
            return self.full.run(stats)

        since = (datetime.fromisoformat(anchor) - timedelta(days=self.settings.incremental_overlap_days)).date()
        stats["since"] = since.isoformat()
        seen: set[int] = set()
        with self.ingestor.Session() as s:
            for name, filters in build_queries(self.client, self.targets, {df: {"from": since.isoformat()}}):
                log.info("incremental → %s: %s", name, filters)
                for rows in self.client.iter_pages(filters, "full"):
                    self.ingestor.ingest_page(s, rows, name, stats, seen, full=True)
        stats["fetched"] = len(seen)
        self.state.set("last_incremental_at", started.isoformat())


class ReconcileStrategy:
    def __init__(self, client: AiesecClient, settings: Settings, targets: Targets,
                 ingestor: Ingestor, state: StateStore, full: FullStrategy):
        self.client, self.settings, self.targets = client, settings, targets
        self.ingestor, self.state, self.full = ingestor, state, full

    def run(self, stats: Counter) -> None:
        started = utcnow()
        with self.ingestor.Session() as s:
            known_ids = set(s.scalars(select(Opportunity.opportunity_id)))
            has_open = s.scalar(
                select(Opportunity.opportunity_id).where(Opportunity.is_open.is_(True)).limit(1)
            )
        if not has_open:
            stats["fallback"] = "full (в базе нет открытых)"
            return self.full.run(stats)

        seen: set[int] = set()
        new_ids: set[int] = set()
        today = date.today()
        with self.ingestor.Session() as s:
            for name, filters in build_queries(self.client, self.targets):
                log.info("reconcile → %s: %s", name, filters)
                for rows in self.client.iter_pages(filters, "slim"):
                    known_rows = []
                    for r in rows:
                        oid = int(r["id"])
                        seen.add(oid)
                        if oid in known_ids:
                            known_rows.append(r)
                        elif self.ingestor.actuality(normalize(r, full=False), today, check_target=False)[0]:
                            new_ids.add(oid)
                    self.ingestor.ingest_page(s, known_rows, name, stats, set(), full=False)
            stats["remote_open"] = len(seen)
            self.ingestor.close_missing(s, seen, stats)
        stats["new_found"] = len(new_ids)

        if new_ids:
            gap_fill(self.client, self.ingestor, new_ids, stats, self.full.run)
        self.state.set("last_reconcile_at", started.isoformat())

def _sha(s: str) -> str:
    return hashlib.sha256(s.encode()).hexdigest()


def mask_token(t: str) -> str:
    return f"…{t[-4:]} (длина {len(t)})" if len(t) > 8 else "…"


def make_persist_tokens(service: SyncService):
    def persist_tokens(c: AiesecClient) -> None:
        """Сохраняет пару токенов одной транзакцией: refresh_token одноразовый."""
        service.set_states({
            "aiesec_token": c.token or "",
            "aiesec_refresh_token": c.refresh_token or "",
            "aiesec_token_expires_at": c.expires_at.isoformat() if c.expires_at else "",
            "aiesec_token_updated_at": utcnow().isoformat(),
        })
    return persist_tokens


def make_init_token(settings: Settings, client: AiesecClient, service: SyncService):
    def init_token() -> None:
        if not (settings.aiesec_token or settings.aiesec_refresh_token):
            log.warning(
                "AIESEC-токен не задан. Сервис поднялся в режиме ожидания. "
                "Загрузите токен: PUT /admin/token (token или refresh_token)"
            )
            return
        env_sha = _sha(f"{settings.aiesec_token or ''}|{settings.aiesec_refresh_token or ''}")
        if service.get_state("token_env_sha") != env_sha:
            service.set_states({
                "token_env_sha": env_sha,
                "aiesec_token": "",
                "aiesec_refresh_token": "",
                "aiesec_token_expires_at": "",
            })
        else:
            access = service.get_state("aiesec_token") or ""
            refresh = service.get_state("aiesec_refresh_token") or ""
            exp = service.get_state("aiesec_token_expires_at") or ""
            if access or refresh:
                client.set_token(
                    access or client.token, source="api",
                    refresh_token=refresh or None,
                    expires_at=datetime.fromisoformat(exp) if exp else None,
                )
                log.info("используются токены, сохранённые в БД (%s)", mask_token(client.token))
        try:
            client.ensure_fresh(refresh_if_unknown=True)
        except AiesecError as e:
            log.error("при старте не удалось обновить токен: %s", e)
    return init_token


def make_token_keepalive(client: AiesecClient):
    def token_keepalive() -> None:
        if client.auth_ok is False:
            return
        try:
            client.ensure_fresh()
        except AiesecError as e:
            log.error("keepalive: %s", e)
    return token_keepalive


def token_status(client: AiesecClient, service: SyncService) -> dict:
    left = client.seconds_left()
    return {
        "masked": mask_token(client.token) if client.token else None,
        "source": client.token_source,
        "auth": {True: "ok", False: "failed", None: "unknown"}[client.auth_ok],
        "auto_refresh": client.can_refresh,
        "expires_in_sec": int(left) if left is not None else None,
        "last_auth_error": client.auth_error,
        "checked_at": client.auth_checked_at,
        "updated_at": service.get_state("aiesec_token_updated_at"),
    }


def clean_token(v: str) -> str:
    return v.strip().strip("\"'").removeprefix("Bearer ").strip()

def _require_schemas(connection) -> None:
    """Схемы создаёт db-init (bootstrap.sql). Здесь — только проверка."""
    schema = settings.db_schema
    if not schema:
        return
    existing = set(inspect(connection).get_schema_names())
    if schema not in existing:
        raise RuntimeError(
            f"Схема '{schema}' не существует в БД. "
            f"Проверьте: docker compose logs db-init. "
            f"Если db-init не отработал — пересоздайте инфраструктуру: "
            f"docker compose down -v && docker compose up -d"
        )

    context.configure(
        url=settings.database_url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
        version_table_schema=settings.db_schema or None,
        include_schemas=True,
    )
    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online() -> None:
    connectable = engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )
    with connectable.connect() as connection:

        connection.commit()

            connection=connection,
            target_metadata=target_metadata,
            version_table_schema=settings.db_schema or None,
            include_schemas=True,
        )
        with context.begin_transaction():

            o.applicants_count,
            o.accepted_count,
            o.date_opened,
            o.applications_close_date,
            o.earliest_start_date,
            o.latest_end_date,
            o.duration AS duration_weeks,
            o.experience_type,
            o.cost_total,
            o.cost_currency,
            o.is_open,
            o.close_reason,
            o.remote_updated_at,
            o.first_seen_at,
            o.last_seen_at,
            o.updated_at
        FROM {SCHEMA}.opportunities o
    """)

    # 4.2 catalog.opportunity_requirement_v1
    op.execute(f"""
        CREATE OR REPLACE VIEW {CATALOG}.opportunity_requirement_v1 AS
        WITH expanded AS (
            SELECT opportunity_id, 'skill' AS kind,
                   jsonb_array_elements(COALESCE(o.skills, '[]'::jsonb)) AS el
            FROM {SCHEMA}.opportunities o
            UNION ALL
            SELECT opportunity_id, 'background',
                   jsonb_array_elements(COALESCE(o.backgrounds, '[]'::jsonb))
            FROM {SCHEMA}.opportunities o
            UNION ALL
            SELECT opportunity_id, 'language',
                   jsonb_array_elements(COALESCE(o.languages, '[]'::jsonb))
            FROM {SCHEMA}.opportunities o
            UNION ALL
            SELECT opportunity_id, 'study_level',
                   jsonb_array_elements(COALESCE(o.study_levels, '[]'::jsonb))
            FROM {SCHEMA}.opportunities o
            UNION ALL
            SELECT opportunity_id, 'nationality',
                   jsonb_array_elements(COALESCE(o.nationalities, '[]'::jsonb))
            FROM {SCHEMA}.opportunities o
            UNION ALL
            SELECT opportunity_id, 'skill_development',
                   jsonb_array_elements(COALESCE(o.skills_developments, '[]'::jsonb))
            FROM {SCHEMA}.opportunities o
        )
        SELECT
            opportunity_id,
            kind,
            (el->>'id')::int          AS constant_id,
            el->>'name'               AS constant_name,
            el->>'parent_name'        AS parent_name,
            el->>'option'             AS option,
            (el->>'level')::int       AS level
        FROM expanded
        WHERE el IS NOT NULL
    """)

    # 4.3 catalog.sync_state_v1
    op.execute(f"""
        CREATE OR REPLACE VIEW {CATALOG}.sync_state_v1 AS
        SELECT
            (SELECT MAX(COALESCE(updated_at, first_seen_at))
               FROM {SCHEMA}.opportunities)                          AS committed_watermark,
            (SELECT MAX(finished_at)
               FROM {SCHEMA}.sync_runs
              WHERE status = 'success')                              AS last_success_at,
            (SELECT COUNT(*) FROM {SCHEMA}.opportunities)            AS rows_total,
            (SELECT COUNT(*) FROM {SCHEMA}.opportunities
              WHERE is_open)                                         AS rows_open,
            COALESCE(
                (SELECT value FROM {SCHEMA}.sync_state
                  WHERE key = 'last_run_kind'),
                (SELECT kind FROM {SCHEMA}.sync_runs
                  ORDER BY started_at DESC LIMIT 1)
            )                                                        AS last_run_kind
    """)
